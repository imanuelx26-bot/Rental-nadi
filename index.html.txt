    function callBackend(action, payload, callback) {

      document.getElementById('camera-title').innerText = type === 'ktp' ? 'Ambil Foto KTP' : 'Ambil Foto Penyewa';
      document.getElementById('modal-camera').classList.remove('hidden');
      
      document.getElementById('camera-video').classList.remove('hidden');
      document.getElementById('camera-captured-img').classList.add('hidden');
      document.getElementById('camera-controls-init').classList.remove('hidden');
      document.getElementById('camera-controls-confirm').classList.add('hidden');

      if (navigator.mediaDevices && navigator.mediaDevices.getUserMedia) {
        navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment' } })
          .then(function(stream) {
            mediaStream = stream;
            document.getElementById('camera-video').srcObject = stream;
          })
          .catch(function(err) {
            showToast("Tidak dapat membuka live kamera: " + err.message + ". Gunakan opsi foto HP!");
            closeCameraModal();
            triggerNativePicker(type);
          });
      } else {
        showToast("Browser ini tidak mendukung live camera. Membuka galeri/kamera HP...");
        closeCameraModal();
        triggerNativePicker(type);
      }
    }

    function closeCameraModal() {
      if (mediaStream) {
        mediaStream.getTracks().forEach(function(track) { track.stop(); });
        mediaStream = null;
      }
      document.getElementById('modal-camera').classList.add('hidden');
    }

    function takePicture() {
      var video = document.getElementById('camera-video');
      var canvas = document.getElementById('camera-canvas');
      canvas.width = video.videoWidth || 640;
      canvas.height = video.videoHeight || 480;
      var ctx = canvas.getContext('2d');
      ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
      
      tempCapturedBase64 = canvas.toDataURL('image/jpeg', 0.8);

      document.getElementById('camera-captured-img').src = tempCapturedBase64;
      document.getElementById('camera-video').classList.add('hidden');
      document.getElementById('camera-captured-img').classList.remove('hidden');

      document.getElementById('camera-controls-init').classList.add('hidden');
      document.getElementById('camera-controls-confirm').classList.remove('hidden');
    }

    function retakePicture() {
      document.getElementById('camera-video').classList.remove('hidden');
      document.getElementById('camera-captured-img').classList.add('hidden');
      document.getElementById('camera-controls-init').classList.remove('hidden');
      document.getElementById('camera-controls-confirm').classList.add('hidden');
    }

    function useCapturedPicture() {
      if (activeCameraType === 'ktp') {
        formKtpBase64 = tempCapturedBase64;
        document.getElementById('img-ktp-preview').src = tempCapturedBase64;
        document.getElementById('img-ktp-preview').classList.remove('hidden');
        document.getElementById('placeholder-ktp').classList.add('hidden');
      } else {
        formPenyewaBase64 = tempCapturedBase64;
        document.getElementById('img-penyewa-preview').src = tempCapturedBase64;
        document.getElementById('img-penyewa-preview').classList.remove('hidden');
        document.getElementById('placeholder-penyewa').classList.add('hidden');
      }
      closeCameraModal();
      showToast("Foto berhasil diambil!");
    }

    // NATIVE FILE / CAMERA INPUT FALLBACK (100% RELIABLE FOR MOBILE)
    function triggerNativePicker(type) {
      if (type === 'ktp') {
        document.getElementById('file-picker-ktp').click();
      } else {
        document.getElementById('file-picker-penyewa').click();
      }
    }

    function handleFileSelect(event, type) {
      var file = event.target.files[0];
      if (!file) return;

      var reader = new FileReader();
      reader.onload = function(e) {
        var base64 = e.target.result;
        if (type === 'ktp') {
          formKtpBase64 = base64;
          document.getElementById('img-ktp-preview').src = base64;
          document.getElementById('img-ktp-preview').classList.remove('hidden');
          document.getElementById('placeholder-ktp').classList.add('hidden');
        } else {
          formPenyewaBase64 = base64;
          document.getElementById('img-penyewa-preview').src = base64;
          document.getElementById('img-penyewa-preview').classList.remove('hidden');
          document.getElementById('placeholder-penyewa').classList.add('hidden');
        }
        showToast("Foto berhasil dimuat!");
      };
      reader.readAsDataURL(file);
    }

    // DASHBOARD DATA LOAD
    function loadDashboardData() {
      callBackend('getDashboardStats', {}, function(res) {
        renderDashboardStats(res);
      });
    }

    function renderDashboardStats(res) {
      document.getElementById('stat-total').innerText = res.totalPenyewa || 0;
      document.getElementById('stat-sewa').innerText = res.sedangSewa || 0;
      document.getElementById('stat-selesai').innerText = res.selesai || 0;
      document.getElementById('stat-today').innerText = res.penyewaanHariIni || 0;

      var listContainer = document.getElementById('dashboard-recent-list');
      if (!res.recentRentals || res.recentRentals.length === 0) {
        listContainer.innerHTML = '<div class="text-center py-4 text-xs text-slate-400">Belum ada penyewaan.</div>';
        return;
      }

      var html = '';
      res.recentRentals.forEach(function(item) {
        var badgeBg = item.status === 'Sedang Sewa' ? 'bg-emerald-100 text-emerald-700 border-emerald-200' : 'bg-blue-100 text-blue-700 border-blue-200';
        html += '<div class="p-3 bg-slate-50 hover:bg-slate-100 rounded-2xl border border-slate-200 flex items-center justify-between cursor-pointer transition" onclick="openDetail(' + escapeJson(item) + ')">';
        html += '<div class="flex items-center space-x-3">';
        html += '<img src="' + (item.fotoPenyewa || 'https://via.placeholder.com/80') + '" class="w-10 h-10 rounded-2xl object-cover border border-slate-200 shadow-sm">';
        html += '<div>';
        html += '<p class="text-xs font-extrabold text-slate-900">' + sanitize(item.namaPenyewa) + '</p>';
        html += '<p class="text-[10px] font-semibold text-slate-500"><span class="text-amber-600">' + sanitize(item.platMotor) + '</span> &bull; ' + sanitize(item.lamaSewa) + '</p>';
        html += '</div>';
        html += '</div>';
        html += '<span class="text-[10px] font-bold px-2 py-1 rounded-lg border ' + badgeBg + '">' + sanitize(item.status) + '</span>';
        html += '</div>';
      });
      listContainer.innerHTML = html;
      lucide.createIcons();
    }

    // PENYEWA LIST LOAD
    function loadPenyewaData() {
      callBackend('getRentals', {
        searchQuery: currentSearch,
        statusFilter: currentFilter,
        page: currentPage,
        limit: 10
      }, function(res) {
        renderPenyewaList(res.data);
      });
    }

    function renderPenyewaList(data) {
      var container = document.getElementById('penyewa-list');
      if (!data || data.length === 0) {
        container.innerHTML = '<div class="text-center py-10 text-xs text-slate-400 font-medium">Tidak ada data penyewa ditemukan.</div>';
        return;
      }

      var html = '';
      data.forEach(function(item) {
        var badgeBg = item.status === 'Sedang Sewa' ? 'bg-emerald-100 text-emerald-700 border-emerald-200' : 'bg-blue-100 text-blue-700 border-blue-200';
        html += '<div class="glass-card p-3 rounded-2xl flex items-center justify-between cursor-pointer hover:border-violet-300 transition" onclick="openDetail(' + escapeJson(item) + ')">';
        html += '<div class="flex items-center space-x-3">';
        html += '<img src="' + (item.fotoPenyewa || 'https://via.placeholder.com/80') + '" class="w-11 h-11 rounded-2xl object-cover border border-slate-200 shadow-sm">';
        html += '<div>';
        html += '<div class="flex items-center space-x-1.5">';
        html += '<h4 class="text-xs font-extrabold text-slate-900">' + sanitize(item.namaPenyewa) + '</h4>';
        html += '<span class="text-[9px] text-violet-700 font-mono font-bold">(' + sanitize(item.idRental) + ')</span>';
        html += '</div>';
        html += '<p class="text-[10px] text-slate-500 mt-0.5">' + sanitize(item.noHp) + ' &bull; <span class="text-amber-600 font-bold">' + sanitize(item.platMotor) + '</span></p>';
        html += '</div>';
        html += '</div>';
        html += '<div class="text-right">';
        html += '<span class="inline-block text-[10px] font-bold px-2 py-0.5 rounded-md border ' + badgeBg + '">' + sanitize(item.status) + '</span>';
        html += '<p class="text-[10px] font-semibold text-slate-500 mt-1">' + sanitize(item.lamaSewa) + '</p>';
        html += '</div>';
        html += '</div>';
      });

      container.innerHTML = html;
      lucide.createIcons();
    }

    function setFilter(status) {
      currentFilter = status;
      var pills = document.querySelectorAll('.filter-pill');
      pills.forEach(function(p) {
        if (p.innerText === status) {
          p.classList.remove('bg-white', 'text-slate-600', 'border', 'border-slate-200');
          p.classList.add('bg-violet-600', 'text-white', 'shadow-sm');
        } else {
          p.classList.remove('bg-violet-600', 'text-white', 'shadow-sm');
          p.classList.add('bg-white', 'text-slate-600', 'border', 'border-slate-200');
        }
      });
      loadPenyewaData();
    }

    function handleSearch() {
      currentSearch = document.getElementById('search-input').value;
      loadPenyewaData();
    }

    // SUBMIT FORM DATA PENYEWA
    function submitForm() {
      var formData = {
        rowIndex: document.getElementById('form-row-index').value,
        idRental: document.getElementById('form-id-rental').value,
        namaPenyewa: document.getElementById('inp-nama').value,
        noHp: document.getElementById('inp-nohp').value,
        lamaSewa: document.getElementById('inp-lama').value,
        bbm: document.getElementById('inp-bbm').value,
        tglJamAmbil: document.getElementById('inp-tgl-ambil').value,
        tglJamKembali: document.getElementById('inp-tgl-kembali').value,
        platMotor: document.getElementById('inp-plat').value,
        biayaSewa: document.getElementById('inp-biaya').value,
        fotoKtp: formKtpBase64 || 'https://via.placeholder.com/300x200?text=KTP',
        fotoPenyewa: formPenyewaBase64 || 'https://via.placeholder.com/200x200?text=Penyewa'
      };

      var btnSubmit = document.getElementById('btn-submit-form');
      btnSubmit.disabled = true;
      btnSubmit.innerText = "Menyimpan...";

      callBackend('saveRental', formData, function(res) {
        btnSubmit.disabled = false;
        btnSubmit.innerText = "Simpan Transaksi";
        if (res.success) {
          showToast(res.message);
          resetForm();
          switchTab('penyewa');
        } else {
          showToast("Gagal: " + res.message);
        }
      });
    }

    function resetForm() {
      document.getElementById('form-penyewa').reset();
      document.getElementById('form-row-index').value = '';
      document.getElementById('form-id-rental').value = '';
      formKtpBase64 = '';
      formPenyewaBase64 = '';
      document.getElementById('img-ktp-preview').classList.add('hidden');
      document.getElementById('placeholder-ktp').classList.remove('hidden');
      document.getElementById('img-penyewa-preview').classList.add('hidden');
      document.getElementById('placeholder-penyewa').classList.remove('hidden');
      nextStep(1);
    }

      document.getElementById('detail-biaya').innerText = "Rp " + Number(item.biayaSewa || 0).toLocaleString('id-ID');

      document.getElementById('detail-img-penyewa').src = item.fotoPenyewa || 'https://via.placeholder.com/200x200';

      if (item.status === 'Sedang Sewa') {
        lblStatus.innerText = "Tandai Selesai";

        lblStatus.innerText = "Kembalikan ke Sedang Sewa";
      }

    function toggleDetailStatus() {
      if (!activeRentalDetail) return;
      var newStatus = activeRentalDetail.status === 'Sedang Sewa' ? 'Selesai' : 'Sedang Sewa';
      activeRentalDetail.status = newStatus;

      showToast("Memperbarui status...");
      closeModalDetail();

      callBackend('updateRentalStatus', {
        idRental: activeRentalDetail.idRental,
        newStatus: newStatus
      }, function(res) {
        showToast(res.message);
        loadPenyewaData();
        loadDashboardData();
      });
    }

    // UTILS
    function escapeJson(obj) {
      return JSON.stringify(obj).replace(/"/g, '&quot;');
    }

    function sanitize(str) {
      if (!str) return '';
      return String(str).replace(/</g, '&lt;').replace(/>/g, '&gt;');
    }

      if (url) window.open(url, '_blank');
